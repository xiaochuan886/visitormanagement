```python
            created_by=user_id
        )
        
        self.db.add(config_entity)
        await self.db.commit()
        await self.db.refresh(config_entity)
        
        return FormConfigurationResponseDTO.from_orm(config_entity)
```

```python
# app/application/services/form_processor_service.py
class FormProcessor:
    """表单处理器"""
    
    async def validate_configuration(self, config: FormConfigurationCreateDTO) -> ValidationResult:
        """验证表单配置"""
        errors = []
        
        # 验证字段定义
        for field in config.fields:
            if not field.get('name'):
                errors.append("字段缺少name属性")
            if not field.get('type'):
                errors.append(f"字段{field.get('name', '未知')}缺少type属性")
        
        # 验证步骤配置
        if config.steps:
            step_fields = set()
            for step in config.steps:
                for field_name in step.get('fields', []):
                    step_fields.add(field_name)
            
            config_fields = {field['name'] for field in config.fields}
            missing_fields = step_fields - config_fields
            if missing_fields:
                errors.append(f"步骤中引用了未定义的字段: {missing_fields}")
        
        return ValidationResult(is_valid=len(errors) == 0, errors=errors)
    
    async def process_form_data(
        self,
        form_data: Dict[str, Any],
        form_config: FormConfigurationModel
    ) -> Dict[str, Any]:
        """处理表单数据"""
        processed_data = {}
        
        for field_config in form_config.fields:
            field_name = field_config['name']
            field_value = form_data.get(field_name)
            
            # 应用验证规则
            if field_config.get('required') and not field_value:
                raise ValueError(f"必填字段{field_name}不能为空")
            
            # 处理字段值
            processed_value = await self._process_field_value(field_value, field_config)
            processed_data[field_name] = processed_value
        
        return processed_data
```

```python  
class WorkflowEngine:
    """工作流引擎"""
    
    async def start_workflow(
        self,
        entity_id: UUID,
        workflow_type: str,
        context: Dict[str, Any]
    ) -> WorkflowExecutionResult:
        """启动工作流"""
        
        # 获取适用的工作流配置
        workflow_config = await self.get_applicable_workflow(entity_id, workflow_type, context)
        
        # 评估触发条件
        if not await self.evaluate_trigger_conditions(entity_id, workflow_config.trigger_conditions, context):
            return WorkflowExecutionResult(
                workflow_id=None,
                initial_status="SKIPPED",
                next_steps=[]
            )
        
        # 尝试自动审批
        auto_approval_result = await self.try_auto_approval(entity_id, workflow_config.auto_approval_rules, context)
        
        if auto_approval_result.approved:
            return WorkflowExecutionResult(
                workflow_id=auto_approval_result.workflow_id,
                initial_status="AUTO_APPROVED",
                next_steps=["notify_approval_success"]
            )
        
        # 启动人工审批流程
        manual_workflow = await self.start_manual_workflow(entity_id, workflow_config, context)
        
        return WorkflowExecutionResult(
            workflow_id=manual_workflow.id,
            initial_status="PENDING_APPROVAL",
            next_steps=manual_workflow.next_steps
        )
```

```python
# app/application/services/spatial_engine_service.py
class SpatialEngine:
    """空间管理引擎"""
    
    async def create_spatial_entity(
        self,
        entity_data: SpatialEntityCreateDTO,
        tenant_id: str,
        user_id: str
    ) -> SpatialEntityResponseDTO:
        """创建空间实体"""
        
        # 验证父实体存在性
        if entity_data.parent_id:
            parent_entity = await self.get_spatial_entity(entity_data.parent_id, tenant_id)
            if not parent_entity:
                raise ValueError("父实体不存在")
            
            # 计算层级路径
            hierarchy_path = f"{parent_entity.hierarchy_path}/{entity_data.entity_code}"
            level_depth = parent_entity.level_depth + 1
        else:
            hierarchy_path = f"/{entity_data.entity_code}"
            level_depth = 0
        
        # 创建空间实体
        spatial_entity = SpatialEntityModel(
            tenant_id=tenant_id,
            entity_code=entity_data.entity_code,
            entity_name=entity_data.entity_name,
            entity_type=entity_data.entity_type,
            parent_id=entity_data.parent_id,
            hierarchy_path=hierarchy_path,
            level_depth=level_depth,
            capacity=entity_data.capacity,
            area_sqm=entity_data.area_sqm,
            coordinates=entity_data.coordinates,
            facilities=entity_data.facilities,
            access_devices=entity_data.access_devices,
            business_rules=entity_data.business_rules,
            access_permissions=entity_data.access_permissions,
            operating_hours=entity_data.operating_hours,
            created_by=user_id
        )
        
        self.db.add(spatial_entity)
        await self.db.commit()
        await self.db.refresh(spatial_entity)
        
        return SpatialEntityResponseDTO.from_orm(spatial_entity)
    
    async def plan_visitor_route(
        self,
        visitor_id: UUID,
        destination_config: Dict[str, Any],
        context: Dict[str, Any]
    ) -> RouteExecutionResult:
        """规划访客路径"""
        
        # 解析目标空间实体
        destination_entity = await self.resolve_spatial_destination(destination_config, context['tenant_id'])
        
        # 确定入口点
        entry_point = await self.determine_entry_point(destination_entity, context)
        
        # 计算路径
        route_path = await self.calculate_route_path(entry_point, destination_entity)
        
        # 分析权限需求
        permission_requirements = await self.analyze_permission_requirements(route_path, context)
        
        # 创建路径记录
        route_entity = VisitorRouteModel(
            visitor_id=visitor_id,
            entry_point_id=entry_point.id,
            destination_id=destination_entity.id,
            route_path=route_path,
            authorized_entities=permission_requirements.authorized,
            restricted_entities=permission_requirements.restricted,
            escort_required_entities=permission_requirements.escort_required,
            tenant_id=context['tenant_id']
        )
        
        self.db.add(route_entity)
        await self.db.commit()
        
        return RouteExecutionResult(
            route_id=route_entity.id,
            route_path=route_path,
            permission_requirements=permission_requirements
        )
```

```typescript
const FormRenderer: React.FC<FormRendererProps> = ({ config, onSubmit, mode = "normal" }) => {
  const [formData, setFormData] = useState({});
  const [currentStep, setCurrentStep] = useState(0);

  const renderField = (field: FormField) => {
    switch (field.type) {
      case 'text':
        return (
          <Input
            key={field.id}
            name={field.name}
            placeholder={field.placeholder}
            required={field.required}
            value={formData[field.name] || ''}
            onChange={(e) => handleFieldChange(field.name, e.target.value)}
          />
        );

            key={field.id}
            name={field.name}
            levels={field.spatialLevels}
            value={formData[field.name]}
            onChange={(value) => handleFieldChange(field.name, value)}
          />
        );
      
      default:
        return null;
    }
  };

  return (
    <div className="form-renderer">
      {config.steps && config.steps.length > 1 && (
        <Steps current={currentStep}>
          {config.steps.map((step, index) => (
            <Steps.Step key={index} title={step.title} />
          ))}
        </Steps>
      )}
      
      <div className="form-content">
        {getCurrentStepFields().map(renderField)}
      </div>
      
      {mode === "normal" && (
        <div className="form-actions">
          {currentStep > 0 && (
            <Button onClick={() => setCurrentStep(currentStep - 1)}>
              上一步
            </Button>
          )}
          
          {currentStep < config.steps.length - 1 ? (
            <Button type="primary" onClick={() => setCurrentStep(currentStep + 1)}>
              下一步
            </Button>
```

```python
# app/core/monitoring.py
import structlog
from prometheus_client import Counter, Histogram, generate_latest

# 监控指标
REQUEST_COUNT = Counter('http_requests_total', 'Total HTTP requests', ['method', 'endpoint', 'status'])
REQUEST_DURATION = Histogram('http_request_duration_seconds', 'HTTP request duration')

logger = structlog.get_logger()

async def monitor_request(request, call_next):
    """请求监控中间件"""
    start_time = time.time()
    
    response = await call_next(request)
    
    duration = time.time() - start_time
    REQUEST_DURATION.observe(duration)
    REQUEST_COUNT.labels(
        method=request.method,
        endpoint=request.url.path,
        status=response.status_code
    ).inc()
    
    logger.info(
        "Request processed",
        method=request.method,
        path=request.url.path,
        status_code=response.status_code,
        duration=duration
    )
    
    return response
```
